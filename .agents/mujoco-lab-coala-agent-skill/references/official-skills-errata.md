```python
import numpy as np, mujoco
spec = mujoco.MjSpec()
spec.compiler.degree = False                      # ângulos do MjSpec em RADIANOS (padrão: graus)
arm = spec.worldbody.add_body(name='arm', pos=[0, 0, 1])
arm.add_geom(type=mujoco.mjtGeom.mjGEOM_CAPSULE, size=[0.05], fromto=[0, 0, 0, 0.5, 0, 0], mass=1)
arm.add_joint(name='shoulder', type=mujoco.mjtJoint.mjJNT_HINGE, axis=[0, 1, 0], range=[-1.57, 1.57])
servo = spec.add_actuator(name='pos_servo', target='shoulder', trntype=mujoco.mjtTrn.mjTRN_JOINT)
servo.set_to_position(kp=200.0, kv=10.0)          # com amortecimento
servo.ctrllimited, servo.ctrlrange = True, [-1.57, 1.57]
rubber = spec.add_default('rubber', spec.default)  # default aplicado NA CRIAÇÃO
rubber.geom.friction = [1.5, 0.01, 0.001]
arm.add_geom(rubber, name='tire', size=[0.02], pos=[0.5, 0, 0])
v = np.array([[0, 0, 0], [1, 0, 0], [0, 1, 0], [0, 0, 1]], np.float32)           # tetraedro (>= 4 vértices)
f = np.array([[0, 2, 1], [0, 1, 3], [0, 3, 2], [1, 2, 3]], np.int32)
spec.add_mesh(name='tet', uservert=v.flatten(), userface=f.flatten())          # listas PLANAS
spec.worldbody.add_body(name='junk', pos=[2, 0, 1]).add_geom(size=[0.1], mass=1)
spec.delete(spec.body('junk'))                     # delete(elemento), não delete_body(nome)
model = spec.compile(); data = mujoco.MjData(model)
assert np.allclose(model.jnt_range[0], [-1.57, 1.57]) and model.nbody == 2 and model.nmesh == 1
assert np.allclose(model.geom_friction[1], [1.5, 0.01, 0.001])

def arm_spec():
    s = mujoco.MjSpec(); b = s.worldbody.add_body(name='base'); b.add_geom(size=[0.05], mass=1)
    h = b.add_body(name='hand', pos=[0, 0, 0.5]); h.add_geom(size=[0.05], mass=1); h.add_joint(name='j', axis=[0, 1, 0]); return s
parent = mujoco.MjSpec(); parent.copy_during_attach = True   # permite reusar o mesmo child e mantê-lo compilável

parent.attach(child, frame=parent.worldbody.add_frame(pos=[1, 0, 0]), prefix='right_')
parent.add_equality(type=mujoco.mjtEq.mjEQ_CONNECT, objtype=mujoco.mjtObj.mjOBJ_BODY,
                    name1='left_hand', name2='right_hand', data=[0.0] * 11)               # sem anchor=; data[0:3] = âncora
assert parent.compile().neq == 1 and child.compile().nbody == 3
```

## Regras para usar as skills oficiais
1. Trate-as como roteiro, não como fonte: **execute o trecho** em `.venv` (cwd privado) antes de adotar — 23 de 63 blocos falham ou têm ressalva.
2.
